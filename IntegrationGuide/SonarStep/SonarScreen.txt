using UnityEngine;

public class SonarScreen : MonoBehaviour
{
    [Header("Scene references (all authored in WynnScene)")]
    public SubmarineMovement movement;
    public VoyageResources resources;
    public HullIntegrity hullIntegrity;
    public OceanSystem ocean;
    public Camera sonarCamera;
    public Transform grid;
    public TextMesh statusText;
    public TextMesh readoutText;
    [Header("View tuning")]
    [Min(20f)] public float horizontalRange = 120f;
    [Range(0.1f, 0.5f)] public float submarineScreenX = 0.2f;
    [Min(1f)] public float gridSpacing = 10f;
    public Color background = new Color(0.002f, 0.025f, 0.016f);
    public Color normalColor = new Color(0.22f, 1f, 0.55f);
    public Color contactColor = new Color(1f, 0.4f, 0.12f);
    [Header("Optional debug overlay")]
    public bool showDebugOverlay = true;
    private int sonarMask;

    private void Awake()
    {
        if (movement == null || sonarCamera == null || sonarCamera.targetTexture == null)
        {
            Debug.LogError("Sonar needs movement, camera and an assigned Render Texture asset.", this);
            enabled = false;
            return;
        }
        sonarMask = sonarCamera.cullingMask;
    }

    private void LateUpdate() => RefreshDisplay();

    public void RefreshDisplay()
    {
        if (movement == null || sonarCamera == null || sonarCamera.targetTexture == null) return;
        bool working = resources == null || resources.SonarWorking;
        RenderTexture target = sonarCamera.targetTexture;
        float aspect = (float)target.width / target.height;
        sonarCamera.aspect = aspect;
        sonarCamera.orthographicSize = Mathf.Max(20f, horizontalRange) / aspect * 0.5f;
        Vector3 position = movement.WorldPosition;
        float lead = horizontalRange * (0.5f - submarineScreenX);
        sonarCamera.transform.position = new Vector3(position.x + lead, position.y + 200f, position.z);
        sonarCamera.transform.rotation = Quaternion.Euler(90f, 0f, 0f);
        sonarCamera.cullingMask = working ? sonarMask : 0;
        sonarCamera.backgroundColor = working ? background : Color.black;
        if (grid != null)
        {
            float spacing = Mathf.Max(1f, gridSpacing);
            grid.position = new Vector3(Mathf.Floor((position.x + lead) / spacing) * spacing, movement.courseOrigin.position.y - 149f, Mathf.Floor(position.z / spacing) * spacing);
            grid.localScale = Vector3.one * (spacing / 10f);
        }
        bool contact = movement.Contact || Time.time - movement.LastContactTime < 0.8f;
        if (statusText != null)
        {
            statusText.color = contact ? contactColor : normalColor;
            statusText.text = !working ? "SONAR OFFLINE / Z: POWER" : contact ? "CONTACT / RISE OR DIVE TO CLEAR" : $"RANGE {horizontalRange:0}m / GRID {gridSpacing:0}m / CONTACTS {movement.CollisionCount}";
        }
        if (readoutText != null)
            readoutText.text = working ? $"DEPTH {movement.Depth:0.0}m\nSPEED {movement.Speed:0.0}m/s\nHDG {movement.Heading:000} DEG" + (ocean!=null && movement.externalState==null ? $"\nGOAL {ocean.goalDepth:0}m" : "") : "";
    }

    private void OnGUI()
    {
        // Host owns instruments and the result UI.
        if(resources!=null && resources.externalState!=null) return;
        if (movement == null) return;
        if(showDebugOverlay) DrawDebugOverlay();
        if (resources != null && resources.Failed)
            GUI.Box(new Rect(Screen.width * 0.5f - 180f, Screen.height * 0.5f - 40f, 360f, 80f), $"VOYAGE FAILED\n{resources.FailureReason}\nExit Play Mode to restart.");
        else if (resources != null && resources.Won)
            GUI.Box(new Rect(Screen.width * 0.5f - 180f, Screen.height * 0.5f - 40f, 360f, 80f), "VOYAGE COMPLETE\nTarget reached. Exit Play Mode to restart.");
    }

    private void DrawDebugOverlay()
    {
        GUI.Box(new Rect(15f, 15f, 340f, 250f), "NAVIGATION TEST");
        GUI.Label(new Rect(30f, 45f, 315f, 25f), $"Gear {movement.gear} | Speed {movement.Speed:0.0} m/s");
        GUI.Label(new Rect(30f, 70f, 315f, 25f), $"Depth {movement.Depth:0.0} m | Distance {movement.Distance:0.0} m");
        GUI.Label(new Rect(30f, 95f, 315f, 25f), $"Heading {movement.Heading:000} | Trim {movement.trim:0.0} | Hits {movement.CollisionCount}");
        if (resources != null)
            GUI.Label(new Rect(30f, 120f, 315f, 25f), $"Battery {resources.Battery:0.0} | Oxygen {resources.Oxygen:0}s");
        if(hullIntegrity!=null)
            GUI.Label(new Rect(30f, 145f, 315f, 25f), $"Hull {hullIntegrity.Hull:0.0}/{hullIntegrity.maximumHull:0.0} | Last damage {hullIntegrity.LastDamage:0.0}");
        GUI.Label(new Rect(30f, 175f, 315f, 25f), "R/F: gear | W/S: rise/dive | Q/E: trim");
        GUI.Label(new Rect(30f, 200f, 315f, 25f), "1/2/3: station | Z: sonar | X: pump");
        GUI.Label(new Rect(30f, 225f, 315f, 25f), "Hold A / D: change horizontal heading");
    }
}
